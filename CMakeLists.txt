cmake_minimum_required(VERSION 3.20)
option(BUILD_FIRMWARE "Build Waveshare RP2350-USB-A firmware" OFF)
if(BUILD_FIRMWARE)
    if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    if(NOT PICO_SDK_PATH)
        message(FATAL_ERROR "Set PICO_SDK_PATH; see docs/development.md")
    endif()
    set(PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/firmware/boards" CACHE STRING "Custom board headers")
    set(PICO_BOARD waveshare_rp2350_usb_a CACHE STRING "Waveshare RP2350-USB-A")
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(keyboard_translator LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(translator STATIC src/core/keyboard.c)
target_include_directories(translator PUBLIC include)
add_library(boot_keyboard STATIC src/input/boot_keyboard.c)
target_include_directories(boot_keyboard PUBLIC include)
add_library(videowriter STATIC examples/videowriter/videowriter.c)
target_include_directories(videowriter PUBLIC examples/videowriter include)
foreach(target translator boot_keyboard videowriter)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wpedantic)
endforeach()
if(BUILD_FIRMWARE)
    pico_sdk_init()
    if(NOT PICO_PIO_USB_PATH AND DEFINED ENV{PICO_PIO_USB_PATH})
        set(PICO_PIO_USB_PATH "$ENV{PICO_PIO_USB_PATH}")
    endif()
    if(NOT EXISTS "${PICO_PIO_USB_PATH}/CMakeLists.txt")
        message(FATAL_ERROR "Set PICO_PIO_USB_PATH to Pico-PIO-USB 0.7.2")
    endif()
    add_subdirectory(${PICO_PIO_USB_PATH} pico-pio-usb)
    add_subdirectory(firmware)
else()
    enable_testing()
    add_executable(translator_tests tests/test_translator.c)
    target_link_libraries(translator_tests PRIVATE translator boot_keyboard videowriter)
    target_compile_options(translator_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
    add_test(NAME translator_tests COMMAND translator_tests)
endif()
